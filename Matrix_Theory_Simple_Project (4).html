"""Supplied output helpers. Students may use these without modification."""
from pathlib import Path
import csv
import numpy as np
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

OUT = Path(__file__).resolve().parent / "results"
OUT.mkdir(exist_ok=True)
plt.rcParams.update({"font.size": 10, "axes.spines.top": False,
                     "axes.spines.right": False, "figure.dpi": 130})


def table(name, fields, rows):
    with (OUT / f"{name}.csv").open("w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(fields)
        writer.writerows(rows)


def figure(name):
    plt.tight_layout()
    plt.savefig(OUT / f"{name}.png", dpi=150)
    plt.close()


def metrics(exact, estimate, tolerance):
    """Absolute, normwise, entrywise and tolerance-scaled errors.

    All inputs are equal-length real vectors. Tolerances must be positive.
    A relative error is NaN if its denominator is zero; never replace it by 0.
    """
    exact = np.asarray(exact, dtype=float)
    estimate = np.asarray(estimate, dtype=float)
    tolerance = np.asarray(tolerance, dtype=float)
    if not (exact.ndim == 1 and exact.shape == estimate.shape == tolerance.shape):
        raise ValueError("Inputs must be equal-length vectors")
    if not all(np.all(np.isfinite(v)) for v in (exact, estimate, tolerance)):
        raise ValueError("Inputs must be finite")
    if not np.all(tolerance > 0):
        raise ValueError("Tolerances must be positive")
    error = np.abs(estimate - exact)
    relative = []
    for order in (1, 2, np.inf):
        scale = np.linalg.norm(exact, order)
        relative.append(np.linalg.norm(error, order) / scale if scale else np.nan)
    entry_relative = np.full(exact.shape, np.nan)
    np.divide(error, np.abs(exact), out=entry_relative, where=exact != 0)
    return {"absolute_inf": float(np.max(error)),
            "relative_1": float(relative[0]), "relative_2": float(relative[1]),
            "relative_inf": float(relative[2]), "entry_relative": entry_relative,
            "worst_entry": float(np.max(entry_relative)),
            "score": float(np.max(error / tolerance))}

import numpy as np

def project(x, u):
    if u @ u == 0:
        raise ValueError("u must be nonzero")
    return (u @ x) / (u @ u) * u


def direction_scan(x, count):
    rows = []
    for theta in np.linspace(0.0, np.pi, count):
        u = np.array([np.cos(theta), np.sin(theta)])
        rows.append([np.degrees(theta), np.linalg.norm(x - project(x, u))])
    return np.array(rows)


def main():

"""Project 3 baseline: eigenvalues, Rayleigh quotient, and stretch.

STUDENT TASKS: extend the matrix cases, refine the direction grid, and test
perturbation amplification. Add 3-5 observation comments with numerical evidence.

import numpy as np
from common import plt, table, figure


def rayleigh(A, x):
    if x @ x == 0:
        raise ValueError("x must be nonzero")
    return (x @ A @ x) / (x @ x)


def scan(B, count):
    rows = []
    for theta in np.linspace(0.0, np.pi, count):
        z = np.array([np.cos(theta), np.sin(theta)])
        rows.append([np.degrees(theta), rayleigh(B, z), np.linalg.norm(B @ z)])
    return np.array(rows)


def main():
    cases = [("symmetric", np.array([[2., 1.], [1., 2.]])),
             ("shear5", np.array([[1., 5.], [0., 1.]]))]
    summary = []
    for name, B in cases:
        rows = scan(B, 181)
        rho = np.max(np.abs(np.linalg.eigvals(B)))
        spectral = np.linalg.norm(B, 2)
        symmetric_part = (B+B.T)/2
        rayleigh_limits = np.linalg.eigvalsh(symmetric_part)
        squared, directions = np.linalg.eigh(B.T @ B)
        v = directions[:, -1]
        assert np.isclose(np.linalg.norm(B @ v), spectral, rtol=1e-10, atol=1e-12)
        summary.append([name, rho, np.linalg.norm(B, 1), spectral,
                        np.linalg.norm(B, np.inf), np.linalg.norm(B, "fro"),
                        np.max(rows[:, 1]), np.max(rows[:, 2]),
                        spectral-np.max(rows[:, 2])])

        for ax in axes:
            ax.set_xlabel("Input direction (degrees)"); ax.legend(fontsize=8)
        fig.suptitle(f"{name}: spectral radius {rho:.3g}; spectral norm {spectral:.3g}")
        figure(f"p3_{name}")

The formulas are supplied. Record observations in comments; no derivation.
"""
import numpy as np
from common import plt, table, figure


def main():
    A = np.array([[2.5, 1.0], [1.0, 1.5]])  # A(t) at t=0.5
    E = np.diag([1.0, -1.0])
    inverse = np.linalg.inv(A)
    reference_inv = -inverse @ E @ inverse
    reference_det = np.linalg.det(A) * np.trace(inverse @ E)
    rows = []
    for h in np.logspace(-1, -10, 19):
        fd_inv = (np.linalg.inv(A+h*E)-np.linalg.inv(A-h*E))/(2*h)
        fd_det = (np.linalg.det(A+h*E)-np.linalg.det(A-h*E))/(2*h)
        rows.append([h, np.linalg.norm(fd_inv-reference_inv, "fro"),
                     abs(fd_det-reference_det)])
    rows = np.array(rows)
    table("optional_derivatives", ["h", "inverse_F_error", "determinant_abs_error"], rows)
